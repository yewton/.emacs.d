=consult-ghostel= は ghostel リポジトリの =extensions/consult-ghostel/= に
同梱される拡張で、=.borgconfig= の =submodule.ghostel.load-path= に
=extensions/consult-ghostel= を追加してロードパスに載せている。
=consult-ghostel-mode= を有効にすると以下が有効になる:

- =consult-ghostel= / =consult-ghostel-project= — ライブプレビュー付きの
  ターミナルバッファピッカー（名前がヒットしなければその名前で新規作成）
- =consult-ghostel-history= — シェルのコマンド履歴から選んで端末へ流し込む
- =consult-buffer= / =consult-project-buffer= の =g= narrow キーで
  ghostel バッファだけに絞り込み、=consult-bookmark= に「Ghostel」グループ
- marginalia のバッファ注釈へターミナルタイトルを前置（marginalia の
  =buffer= / =project-buffer= 注釈が優先されてタイトルが隠れる問題を、
  拡張側が =marginalia-annotators= 経由で解決する）

marginalia のタイトル前置は =consult-ghostel-mode= の有効化時に登録されるので、
=marginalia= を先に =require= し、そのあとでモードを有効にする。

#+begin_src emacs-lisp
(require 'ghostel)
(require 'marginalia)
(require 'consult-ghostel)
(consult-ghostel-mode 1)
(require 'spinner)
(require 'toncs-config-terminal)

(defvar toncs-ghostel-terminal-order nil
  "Terminal buffers in first registration order, retained across mode changes.")

(defun toncs-ghostel-unregister-terminal ()
  "Remove the current buffer from terminal registration order."
  (setq toncs-ghostel-terminal-order
        (delq (current-buffer) toncs-ghostel-terminal-order)))

(put 'toncs-ghostel-unregister-terminal 'permanent-local-hook t)

(defun toncs-ghostel-register-terminal (&optional buffer)
  "Append BUFFER to registration order once and install its cleanup hook.
BUFFER defaults to the current buffer."
  (with-current-buffer (or buffer (current-buffer))
    (unless (memq (current-buffer) toncs-ghostel-terminal-order)
      (setq toncs-ghostel-terminal-order
            (append toncs-ghostel-terminal-order (list (current-buffer)))))
    (add-hook 'kill-buffer-hook #'toncs-ghostel-unregister-terminal nil t)))

(defun toncs-config-ghostel-configure ()
  (add-hook 'ghostel-mode-hook #'toncs-ghostel-register-terminal)
  (add-hook 'ghostel-mode-hook #'toncs-terminal-mode-setup))
#+end_src
