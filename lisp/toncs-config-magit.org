#+begin_src emacs-lisp
(defvar auto-compile-on-load-mode)
(defvar forge-database-connector)

(eval-and-compile
  ;; Magit and Forge provide features before finishing their cyclic loads.
  ;; Recompiling their dependencies at that point sees incomplete EIEIO
  ;; classes and Transient definitions. Borg builds these packages together;
  ;; load-prefer-newer still selects updated sources when bytecode is stale.
  (let ((auto-compile-on-load-mode nil))
    (require 'magit)
    ;; Set this before loading Forge, including during byte compilation.
    (setq forge-database-connector
          (if (version< emacs-version "29") 'sqlite-module 'sqlite-builtin))
    (require 'forge)))

(defun toncs-config-magit-configure ()
  (setq magit-diff-refine-hunk 'all)
  (setq magit-display-buffer-function 'magit-display-buffer-fullframe-status-v1)
  ;; https://emacsmirror.net/manual/borg/Bootstrapping-from-scratch.html
  (magit-add-section-hook 'magit-status-sections-hook
                          'magit-insert-modules
                          'magit-insert-stashes
                          'append))
#+end_src
